  function hideLoader(){
    const loader = document.getElementById('loader');
    if (loader) loader.classList.add('hide');
  }
  let loaderHidden = false;
  function hideLoaderOnce(){
    if (loaderHidden) return;
    loaderHidden = true;
    hideLoader();
  }
  document.addEventListener('DOMContentLoaded', () => {
    setTimeout(hideLoaderOnce, 300);
  });
  setTimeout(hideLoaderOnce, 1200);

  document.addEventListener('DOMContentLoaded', () => {
    const themeInput = document.getElementById('themeInput');
    const root = document.documentElement;

    const savedTheme = localStorage.getItem('weblify-theme');
    if (savedTheme === 'dark'){
      root.setAttribute('data-theme','dark');
      themeInput.checked = true;
    }
    themeInput.addEventListener('change', () => {
      const theme = themeInput.checked ? 'dark' : 'light';
      root.setAttribute('data-theme', theme);
      localStorage.setItem('weblify-theme', theme);
    });

    const revealEls = document.querySelectorAll('.reveal');
    const io = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting){
          entry.target.classList.add('in');
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.15, rootMargin: '0px 0px -40px 0px' });
    revealEls.forEach(el => io.observe(el));

    const spyLinks = document.querySelectorAll('.cir-tabs a.cir-tabs__t');
    const spySections = ['home','services','automation','custom-work','contact'];
    const spyIo = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting){
          spyLinks.forEach(l => l.classList.toggle('active', l.dataset.section === entry.target.id));
        }
      });
    }, { rootMargin: '-45% 0px -45% 0px' });
    spySections.forEach(id => {
      const el = document.getElementById(id);
      if (el) spyIo.observe(el);
    });
    spyLinks.forEach(l => {
      l.addEventListener('click', e => {
        e.preventDefault();
        const el = document.getElementById(l.dataset.section);
        if (el) el.scrollIntoView({ behavior: 'smooth' });
      });
    });

    document.querySelectorAll('.faq-item').forEach(item => {
      item.addEventListener('click', () => {
        const wasOpen = item.classList.contains('open');
        document.querySelectorAll('.faq-item').forEach(i => i.classList.remove('open'));
        if (!wasOpen) item.classList.add('open');
      });
    });

    const quoteForm = document.getElementById('quoteForm');
    if (quoteForm){
      quoteForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const name = document.getElementById('qName').value.trim();
        const email = document.getElementById('qEmail').value.trim();
        const message = document.getElementById('qMessage').value.trim();

        const subject = `Quote Request — ${name || 'New enquiry'}`;
        const body =
          `Name: ${name}\n` +
          `Email: ${email}\n\n` +
          `What they want to do:\n${message}`;

        const mailto = `mailto:business.weblify@gmail.com?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`;
        window.location.href = mailto;
      });
    }
  });
</script>

<script type="module">
  import { createTimeline, stagger, splitText, animate, onScroll } from 'https://esm.sh/animejs';

  function initWordmark(){
    const { words, chars } = splitText('.hero-word p', {
      words: { wrap: 'clip' },
      chars: true,
    });
    createTimeline({
      defaults: { ease: 'inOut(3)', duration: 650 }
    })
    .add(words, {
      y: [$el => +$el.dataset.line % 2 ? '100%' : '-100%', '0%'],
    }, stagger(125))
    .add(chars, {
      y: $el => +$el.dataset.line % 2 ? '100%' : '-100%',
    }, stagger(10, { from: 'random' }))
    .init();
  }

  function initChartDraw(){
    const section = document.getElementById('automation');
    if (!section) return;
    animate('.js-draw', {
      strokeDashoffset: [600, 0],
      duration: 1400,
      easing: 'easeOutQuad',
      autoplay: onScroll({ target: section, debug: false })
    });
  }

  try { initWordmark(); } catch(e) { console.warn('wordmark animation failed', e); }
  try { initChartDraw(); } catch(e) { console.warn('chart draw animation failed', e); }
</script>
